Installation Of Terraform:
To install Terraform follow the below steps.
-> Lunch ubuntu18.04 version
-> Run the below commands
[] curl -fsSL https://apt.releases.hashicorp.com/gpg | sudo apt-key add -
[] sudo apt-add-repository "deb [arch=amd64] https://apt.releases.hashicorp.com $(lsb_release -cs) main"
[] sudo apt-get update && sudo apt-get install terraform
-> Check the version of the terraform
[] terraform version
-> Download secret key and access key from AWS Dashboard > Click on Account on the right sidebar On the accounts's detail page, click the 'Security credentials' tab. Scroll down to the 'Access keys' section. > Click 'Create access key'. > We'll see our Access Key ID and Secret Access Key on screen. Click 'Download .csv file' to save both keys
-> Create a directory and go inside directory
[] mkdir scripts
[] cd scripts

#Script 5: AWS Resource Creation
[] vi main.tf
# Create new security group :
resource "aws_security_group" "Any Name" {
    name = "Any Name security group"
    vpc_id = var.vpc.id

    ingress {
        description = "inbound"
        from_port = 22
        to_port = 22
        protocol = "tcp"
        cidr_blocks = [var.vpc.cidr]
    }

    egress {
        description = "outbound"
        from_port = 0
        to_port = 0
        protocol = "tcp"
        cidr_blocks = [var.vpc.cidr]
    }
    
    tags = {
        sg_name = "Any Name-security-group"
    }

}

# Create EC2 instance
resource "aws_instance" "ec2_instance" {
    ami = ""
    instance_type = ""
    vpc_security_group_ids = [aws_security_group.AnyName.id]
    tags = {Name = "HTTP_SERVER"}
    key_name = ""
    count = 2
}

[] vi provider.tf
# Provider :

    region = var.AWS_REGION
}

  default     = "us-east-1"
}

  default = {
    id = "VPC I'D"
    cidr = "0.0.0.0/0"

#Script 6: AWS Resource Creation
[] vi main.tf
# Create vpc :
resource "aws_vpc" "MyVPC" {
  cidr_block = var.My_VPC.cidr
  tags = {
    Name = var.My_VPC.vpc_name
  }
}

# Create dedicated Subnet for your vpc :
resource "aws_subnet" "MySubnet" {
  vpc_id                  = aws_vpc.MyVPC.id
  cidr_block              = var.My_VPC.subnet_cidr
  map_public_ip_on_launch = "true"
  availability_zone       = "us-east-1b"
  tags = {
    Name = var.My_VPC.subnet_name
  }
}

# Create instance :
resource "aws_instance" "MyEC2" {
  ami = var.My_Instance.myimage
  instance_type = ""
  subnet_id = aws_subnet.MySubnet.id
  tags = {
    Name = var.My_Instance.mytag
  }
}

#Script 7: AWS Resource Creation
[] vi main.tf
# Create EBS volume
resource "aws_ebs_volume" "Storage_Vol" {
  availability_zone = "us-east-1a"
  size              = 1
  tags = {
    Name = var.Instance_Data.name
  }
}

# Create new instance
resource "aws_instance" "Web" {
  ami               = var.Instance_Data.ami
  availability_zone = "us-east-1a"
  instance_type     = var.Instance_Data.type
  tags = {
    Name = var.Instance_Data.name
  }
}

# Attach newly create EBS volume to instance
resource "aws_volume_attachment" "EBS_Att" {
  device_name = var.Instance_Data.drive
  volume_id   = aws_ebs_volume.Storage_Vol.id
  instance_id = aws_instance.Web.id
}

    region = var.AWS_REGION
}

# Region

  default     = "us-east-1"
}

# Image

  default = {
    ami = ""
    type = "Instance Type"
    name = "EBS_Server"
    drive = "/dev/sde"
  }
}

[] terraform output >> file.txt: You can do terraform output any point in time even after apply which will print the results.
[] terraform output public_ip: You can extract the output to a file.
[] terraform output ip_address >> /etc/ansible/hosts: You can also fetch outputs separately by calling the name.
